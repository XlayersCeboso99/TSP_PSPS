@startuml
autonumber

actor Estudiante as E
participant "InterfazUI" as UI
participant "Pregunta" as P
participant "Subtema" as S
participant "Tema" as T
participant "Módulo" as M
participant "EstadoLocal" as EL


E -> UI : Selecciona una respuesta

UI -> P : evaluarRespuesta(respuesta)

activate P
P -> P : determinarCorrectaIncorrecta()
P -> P : marcarRespondida()
P --> UI : resultado + retroalimentación
deactivate P


UI -> S : actualizarProgreso()

activate S
S -> S : verificarCompletitud()

alt Todas las preguntas han sido respondidas

    S -> S : establecerEstado(COMPLETADO)

    S -> T : notificarSubtemaCompletado()

    activate T
    T -> T : verificarCompletitud()

    alt Todos los subtemas están completados

        T -> T : setCompletado(true)

        T -> M : notificarTemaCompletado()

        activate M
        M -> M : verificarCompletitud()

        alt Todos los temas están completados
            M -> M : setCompletado(true)
        end

        deactivate M
    end

    deactivate T

else Quedan preguntas pendientes

    S -> S : establecerEstado(EN_PROGRESO)

end

deactivate S


UI -> EL : guardarEstadoOffline()

activate EL
EL --> UI : Estado guardado
deactivate EL


UI --> E : Mostrar correcta/incorrecta
UI --> E : Mostrar retroalimentación
UI --> E : Actualizar progreso visual

@enduml
